===== Завдання 2. Node.js HTTP-додаток =====
$ find node_http_app -type f | sort

node_http_app/app.js

$ cat node_http_app/app.js
const express = require('express');
const app = express();

app.get('/', (req, res) => res.send('Hello from Docker!'));
app.get('/animal', (req, res) => res.send('Elephant'));

app.listen(3000, () => console.log('Server is running on port 3000'));

$ cat node_http_app/Dockerfile
FROM node:alpine
WORKDIR /app
COPY package*.json ./
RUN npm install
COPY . .
EXPOSE 3000
CMD ["node", "app.js"]

===== Завдання 3. Node.js і змінна оточення =====
$ find node_env_app -type f | sort
node_env_app/Dockerfile
node_env_app/app.js
node_env_app/package.json

$ cat node_env_app/app.js
const express = require('express');
const app = express();

app.get('/', (req, res) => res.send('Welcome to my Docker app'));
app.get('/config', (req, res) => res.send(process.env.APP_MODE));

app.listen(3000, () => console.log('Server is running on port 3000'));

$ cat node_env_app/Dockerfile
FROM node:alpine
WORKDIR /app
COPY package*.json ./
RUN npm install
COPY . .
ENV APP_MODE=development
EXPOSE 3000
CMD ["node", "app.js"]
